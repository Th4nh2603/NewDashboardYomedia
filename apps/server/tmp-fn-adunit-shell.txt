function getBannerAdUnit(ad_view) {
    $.get('/banner/listAdUnits/'+ ad_view, function (data) {
        $('#optAdUnits').html(data.html);
        $("#optAdUnits option").trigger('chosen:updated');
    }).fail(function (jqXhr){
        if (jqXhr.status === 401) {
            $(location).prop('pathname', 'auth/login');
        }
    });
}

function putBannerClone(selId) {
    var token = $('input[name="_token"]').val();
    $.ajax({
        url: 'banner/clone',
        method: 'put',
        data: {_token: token, id: selId}
    }).done(function (data) {
        showNotification(data.message, 'success');
        $("#ListGrid").trigger('reloadGrid');
    }).fail(function (jqXhr) {
        if (jqXhr.status === 401) {
            $(location).prop('pathname', 'auth/login');
        }
        if (jqXhr.status === 422) {
            var errors = jqXhr.responseJSON;
            showNotification(errors.message, 'error');
        }
    });
}
